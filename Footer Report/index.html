<script>
  const form = document.getElementById('subscription-form');
  const responseBox = document.getElementById('form-response');

  form.addEventListener('submit', async function (e) {
    e.preventDefault();

    const formData = new FormData(form);

    try {
      const response = await fetch(form.action, {
        method: 'POST',
        body: formData
      });

      const data = await response.json();

      responseBox.style.display = 'block';

      if (response.ok) {
        if (data.alreadyRegistered) {
          responseBox.textContent = 'You are already registered.';
          responseBox.style.color = 'orange';
        } else {
          responseBox.textContent = 'Thank you! You have successfully subscribed. Please confirm your email.';
          responseBox.style.color = 'green';
        }
      } else {
        responseBox.textContent = 'Something went wrong. Please try again.';
        responseBox.style.color = 'red';
      }

    } catch (error) {
      responseBox.style.display = 'block';
      responseBox.textContent = 'Network error. Please try again.';
      responseBox.style.color = 'red';
    }
  });
</script>
